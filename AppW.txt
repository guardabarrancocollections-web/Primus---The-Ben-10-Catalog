};

const STATUS_ORDER = [
  STATUSES.MISSING,
  STATUSES.NEW_IN_BOX,
  STATUSES.LOOSE_COMPLETE,
  STATUSES.LOOSE_INCOMPLETE,

export default function App() {
  const [collectionState, setCollectionState] = useState({});
  const [currentSeriesIndex, setCurrentSeriesIndex] = useState(0);
  const [isMenuOpen, setIsMenuOpen] = useState(false);
  const [isProfileOpen, setIsProfileOpen] = useState(false);
  const [isOnline, setIsOnline] = useState(true);

  // Toggle Figure Status
  const toggleStatus = (figureId) => {
    setCollectionState((prevState) => {
      const currentStatus = prevState[figureId] || STATUSES.MISSING;
      const currentIndex = STATUS_ORDER.indexOf(currentStatus);
      const nextIndex = (currentIndex + 1) % STATUS_ORDER.length;
      return {
        ...prevState,
        [figureId]: STATUS_ORDER[nextIndex],
      };
    });
  };

  const activeSeries = LOCAL_SERIES_DATA[currentSeriesIndex] || LOCAL_SERIES_DATA[0];

  // Helper function to format items into rows of 3 to avoid layout breaking in FlatList/SectionList grids
  const formatRows = (data) => {
    const rows = [];
    for (let i = 0; i < data.length; i += 3) {
      rows.push({
        id: `row-${i}`,
        items: data.slice(i, i + 3),
      });
    }
    return rows;
  };

  const sections = LOCAL_SERIES_DATA.map((series, idx) => ({
    ...series,
    seriesIndex: idx,
    data: formatRows(series.data),
  }));

  const renderRow = ({ item }) => {
    return (
      <View style={styles.rowWrapper}>
        {item.items.map((fig) => {
          const status = collectionState[fig.id] || STATUSES.MISSING;
          let imageSource = fig.boxedImageUrl;
          let isDimmed = false;

          if (status === STATUSES.MISSING) {
            imageSource = fig.boxedImageUrl;
            isDimmed = true;
          } else if (status === STATUSES.NEW_IN_BOX) {
            imageSource = fig.boxedImageUrl;
          } else if (status === STATUSES.LOOSE_COMPLETE || status === STATUSES.LOOSE_INCOMPLETE) {
            imageSource = fig.looseImageUrl;
          }

          const getStatusStyle = () => {
            switch (status) {
              case STATUSES.NEW_IN_BOX:
                return { bg: '#2ECC71', text: 'NEW IN BOX' };
              case STATUSES.LOOSE_COMPLETE:
                return { bg: '#10B981', text: 'LOOSE - COMPLETE' };
              case STATUSES.LOOSE_INCOMPLETE:
                return { bg: '#2563EB', text: 'LOOSE - INCOMPLETE' };
              default:
                return { bg: '#374151', text: 'MISSING' };
            }
          };

          const statusStyle = getStatusStyle();

          return (
            <View key={fig.id} style={styles.figureCard}>
              <View style={styles.imageContainer}>
                <Image
                  source={imageSource}
                  style={[styles.figureImage, isDimmed && styles.dimmedImage]}
                  resizeMode="contain"
                />
              </View>
              <TouchableOpacity
                style={[styles.statusButton, { backgroundColor: statusStyle.bg }]}
                onPress={() => toggleStatus(fig.id)}
                activeOpacity={0.8}
              >
                <Text style={styles.statusText} numberOfLines={1}>
                  {statusStyle.text}
                </Text>
              </TouchableOpacity>
            </View>
          );
        })}
        {/* Placeholder cells to fill last row if fewer than 3 items */}
        {Array.from({ length: 3 - item.items.length }).map((_, padIdx) => (
          <View key={`pad-${padIdx}`} style={styles.emptyCard} />
        ))}
      </View>
    );
  };

  return (
    <SafeAreaView style={styles.container}>
      <StatusBar barStyle="dark-content" />
      <ImageBackground
        source={activeSeries.backgroundUrl}
        style={styles.backgroundImage}
        resizeMode="cover"
      >
        {/* Fixed App Bar with Centered Logo */}
        <View style={styles.appBar}>
          <TouchableOpacity
            style={styles.iconButton}
            onPress={() => setIsMenuOpen(true)}
            activeOpacity={0.7}
          >
            <Text style={styles.menuIcon}>☰</Text>
          </TouchableOpacity>

          <View style={styles.logoCenterWrapper} pointerEvents="none">
            <Image source={activeSeries.logoUrl} style={styles.seriesLogo} resizeMode="contain" />
          </View>

          <View style={styles.rightIcons}>
            <TouchableOpacity
              style={styles.iconButton}
              onPress={() => Alert.alert('Search', 'this will implemented soon')}
              activeOpacity={0.7}
            >
              <Text style={styles.actionIcon}>🔍</Text>
            </TouchableOpacity>
            <TouchableOpacity
              style={styles.iconButton}
              onPress={() => setIsProfileOpen(true)}
              activeOpacity={0.7}
            >
              <Text style={styles.actionIcon}>👤</Text>
            </TouchableOpacity>
          </View>
        </View>

        {/* Structured Grid via SectionList */}
        <SectionList
          sections={sections}
          keyExtractor={(item) => item.id}
          renderItem={renderRow}
          renderSectionHeader={({ section }) => (
            <View style={styles.sectionHeaderContainer}>
              <Text style={styles.sectionTitle}>{section.title}</Text>
              <Text style={styles.sectionYears}>{section.years}</Text>
            </View>
          )}
          contentContainerStyle={styles.listContent}
          onViewableItemsChanged={({ viewableItems }) => {
            if (viewableItems.length > 0) {
              const currentSec = viewableItems[0].section;
              if (currentSec && currentSec.seriesIndex !== undefined) {
                setCurrentSeriesIndex(currentSec.seriesIndex);
              }
            }
          }}
          viewabilityConfig={{ itemVisiblePercentThreshold: 10 }}
        />

        {/* Hamburger Menu Modal */}
        <Modal
          visible={isMenuOpen}
          transparent={true}
          animationType="fade"
          onRequestClose={() => setIsMenuOpen(false)}
        >
          <TouchableOpacity
            style={styles.modalOverlay}
            activeOpacity={1}
            onPress={() => setIsMenuOpen(false)}
          >
            <View style={styles.drawerContainer}>
              <Text style={styles.drawerTitle}>Navigation</Text>
              <TouchableOpacity
                style={styles.drawerItem}
                onPress={() => setIsMenuOpen(false)}
              >
                <Text style={styles.drawerItemText}>📰 News</Text>
              </TouchableOpacity>
              <TouchableOpacity
                style={styles.drawerItem}
                onPress={() => setIsMenuOpen(false)}
              >
                <Text style={styles.drawerItemText}>🏆 Leaderboard</Text>
              </TouchableOpacity>
              <TouchableOpacity
                style={styles.drawerItem}
                onPress={() => setIsMenuOpen(false)}
              >
                <Text style={styles.drawerItemText}>🔍 Filter</Text>
              </TouchableOpacity>
            </View>
          </TouchableOpacity>
        </Modal>

        {/* Profile Modal */}
        <Modal
          visible={isProfileOpen}
          transparent={true}
          animationType="slide"
          onRequestClose={() => setIsProfileOpen(false)}
        >
          <TouchableOpacity
            style={styles.modalOverlay}
            activeOpacity={1}
            onPress={() => setIsProfileOpen(false)}
          >
            <View style={styles.profileCardModal}>
              <View style={styles.profileHeaderRow}>
                <Text style={styles.profileTitle}>Collector Profile</Text>
                <View style={styles.statusBadge}>
                  <View
                    style={[
                      styles.statusDot,
                      { backgroundColor: isOnline ? '#10B981' : '#6B7280' },
                    ]}
                  />
                  <Text style={styles.statusTextBadge}>
                    {isOnline ? 'Online' : 'Offline'}
                  </Text>
                </View>
              </View>

              <TouchableOpacity
                style={styles.profileOptionButton}
                onPress={() => Alert.alert('Account', 'Log in / Register selected')}
              >
                <Text style={styles.profileOptionText}>🔑 Log in / Register</Text>
              </TouchableOpacity>

              <TouchableOpacity
                style={styles.profileOptionButton}
                onPress={() => Alert.alert('Score', 'Updating Leaderboard Score...')}
              >
                <Text style={styles.profileOptionText}>📊 Update Leaderboard Score</Text>
              </TouchableOpacity>

              <TouchableOpacity
                style={styles.profileOptionButton}
                onPress={() => Alert.alert('Settings', 'Settings opened')}
              >
                <Text style={styles.profileOptionText}>⚙️ Settings</Text>
              </TouchableOpacity>

              <TouchableOpacity
                style={styles.closeButton}
                onPress={() => setIsProfileOpen(false)}
              >
                <Text style={styles.closeButtonText}>Close</Text>
              </TouchableOpacity>
            </View>
          </TouchableOpacity>
        </Modal>
      </ImageBackground>
    </SafeAreaView>
  );
}
